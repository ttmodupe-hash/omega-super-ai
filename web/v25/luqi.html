import os import json import asyncio from typing import Dict from fastapi import FastAPI, WebSocket, WebSocketDisconnect from fastapi.middleware.cors import CORSMiddleware from dotenv import load_dotenv # LLM SDK Imports from openai import AsyncOpenAI import anthropic load_dotenv() app = FastAPI(title="Luqi AI Agent Engine", version="25.5") # Enable CORS for local testing/frontend serving app.add_middleware( CORSMiddleware, allow_origins=["*"], allow_credentials=True, allow_methods=["*"], allow_headers=["*"], ) # Initialize LLM Async Clients openai_client = AsyncOpenAI(api_key=os.getenv("os.getenv()")) anthropic_client = anthropic.AsyncAnthropic(api_key=os.getenv("os.environ")) # Store active connection sessions class ConnectionManager: def __init__(self): self.active_connections: Dict[str, WebSocket] = {} async def connect(self, session_id: str, websocket: WebSocket): await websocket.accept() self.active_connections[session_id] = websocket def disconnect(self, session_id: str): if session_id in self.active_connections: del self.active_connections[session_id] async def send_token(self, websocket: WebSocket, token: str): await websocket.send_text(json.dumps({"type": "token", "content": token})) async def send_end(self, websocket: WebSocket): await websocket.send_text(json.dumps({"type": "end"})) manager = ConnectionManager() # --- Stream Generator: OpenAI GPT-4o --- async def stream_openai_tokens(prompt: str, websocket: WebSocket): try: response = await openai_client.chat.completions.create( model="gpt-4o", messages=[ {"role": "system", "content": "You are LUQI AI, an advanced autonomous AI assistant."}, {"role": "user", "content": prompt} ], stream=True, ) async for chunk in response: if chunk.choices and chunk.choices[0].delta.content: token = chunk.choices[0].delta.content await manager.send_token(websocket, token) except Exception as e: await manager.send_token(websocket, f"\n[OpenAI Stream Error: {str(e)}]") # --- Stream Generator: Anthropic Claude 3.5 Sonnet --- async def stream_anthropic_tokens(prompt: str, websocket: WebSocket): try: async with anthropic_client.messages.stream( model="claude-3-5-sonnet-20241022", max_tokens=1024, system="You are LUQI AI, an advanced autonomous AI assistant.", messages=[{"role": "user", "content": prompt}] ) as stream: async for text_delta in stream.text_stream: await manager.send_token(websocket, text_delta) except Exception as e: await manager.send_token(websocket, f"\n[Anthropic Stream Error: {str(e)}]") # --- WebSocket Endpoint --- @app.websocket("/api/v25/luqi/ws") async def websocket_endpoint(websocket: WebSocket, session_id: str = "default"): await manager.connect(session_id, websocket) print(f"Client connected: {session_id}") try: while True: # Receive text prompt from UI frontend data_raw = await websocket.receive_text() data = json.loads(data_raw) user_message = data.get("message", "") provider = data.get("provider", "openai") # Default to openai, or 'anthropic' if user_message: if provider == "anthropic": await stream_anthropic_tokens(user_message, websocket) else: await stream_openai_tokens(user_message, websocket) # Signal completion to UI cursor await manager.send_end(websocket) except WebSocketDisconnect: manager.disconnect(session_id) print(f"Client disconnected: {session_id}") except Exception as e: print(f"WebSocket execution error: {e}") manager.disconnect(session_id) # --- REST API Compatibility Routes --- @app.get("/api/v25/luqi/tools") async def get_tools(): return { "tools": [ {"name": "web_search", "description": "Execute live web searches"}, {"name": "code_interpreter", "description": "Execute Python code safely"}, {"name": "memory_store", "description": "Persist long-term facts"} ] } if __name__ == "__main__": import uvicorn uvicorn.run("server:app", host="0.0.0.0", port=8000, reload=True)
